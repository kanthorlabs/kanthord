# Council Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `kanthord council "<idea>" --out <dir>`. It triages an idea, asks role agents on their own models in isolated workbench sessions, synthesizes a direction and writes Mission Service plan files plus a `council.md` record.

**Architecture:** The command is a client of existing server operations only: `agent.enablement.get` and the `workbench.session.*` operations. The orchestration is a pure function over an injected `Ask` callback, so every step is tested with no server. Plan files are rendered by the Mission Service serializer and checked by the Mission Service parser.

**Tech Stack:** Node 24 (native TypeScript type stripping), TypeScript, zod 4.4.3, commander, `node:test`, pnpm 11.24.0.

**Spec:** `docs/brainstorm/specs/2026-10-09-council-design.md` (root repository). Read it before Task 1.

## Global Constraints

- All engine code lives in the `engine` submodule on its own branch `feat/council`. Paths below are relative to `engine/` unless they start with `docs/brainstorm/`.
- Every name outside code is snake_case. JSON keys of every answer schema are snake_case.
- Error codes are exactly: `cli.council.token_required`, `cli.council.agent_not_enabled`, `cli.council.output_directory_not_empty`, `cli.council.agent_session_failed`, `cli.council.answer_invalid`, `cli.council.plan_file_invalid`.
- Agent names are exactly: `council@1`, `po@1`, `designer@1`, `user@1`, `architect@1`, `security@1`. No agent name equals a worker name.
- Complexity to roles and rounds: `S` → `po@1`, `user@1`, 1 round. `M` → `po@1`, `user@1`, `designer@1`, `architect@1`, 1 round. `L` → all five roles, 2 rounds.
- A role session receives only the idea, the triage result and the questions of its round. Never another role's answer.
- The command writes nothing to `<dir>` unless the whole run succeeds.
- A session that does not end within 10 minutes fails.
- Prompt files hold only tab, newline and printable characters (test in `src/agent/catalog.test.ts`). Write them in ASCII.
- pnpm forwards `--` literally: `pnpm run X --flag`.
- Commit subjects use `<type>(<scope>): <subject>` in the present tense. Stage exact paths only. Never `git add -A`.

## Review Focus

- A model wraps its JSON in prose, or writes several fenced blocks: the parser takes the last ` ```json ` block, else the whole trimmed text (test in Task 3).
- A model returns text with leading or trailing blank lines, or YAML-special characters in a verification: rendering trims text fields, and the round trip through the Mission parser still holds (test in Task 7).
- A model returns two entries with one filename, a task with `depends_on`, or a `parent` that names no file of the set: `cli.council.plan_file_invalid`, nothing written (tests in Task 7).
- `--out` names an existing regular file instead of a directory: `cli.council.output_directory_not_empty`, checked before any model call (test in Task 7).
- The server answers a failure or an indeterminate result in the middle of a run: `cli.council.agent_session_failed`, the run is aborted and nothing is written (tests in Task 5).

```ts
export function serializePlanFile(entry: ImportEntry): string {
  const front = stringify({
    ...(entry.id === undefined ? {} : { id: entry.id }),
    kind: entry.kind,
    ...(entry.parent === undefined ? {} : { parent: entry.parent }),
    ...(entry.kind === NodeKind.Task
      ? {}
      : { depends_on: entry.depends_on ?? [] }),
    bindings: entry.bindings,
    verifications: entry.verifications,
  });
  return `---\n${front}---\n# ${entry.name}\n\n## Requirement\n\n${entry.requirement}\n\n## Criterion\n\n${entry.criterion}\n`;
}
```

```ts
export const complexitySchema = z.enum(Complexity);
export type Complexity = z.infer<typeof complexitySchema>;

export const COUNCIL_AGENT = "council@1";
export const RoleAgent = {
  Po: "po@1",
  User: "user@1",
  Designer: "designer@1",
  Architect: "architect@1",
  Security: "security@1",

  Record<Complexity, readonly RoleAgent[]>
> = {
  S: [RoleAgent.Po, RoleAgent.User],
  M: [RoleAgent.Po, RoleAgent.User, RoleAgent.Designer, RoleAgent.Architect],
  L: [
    RoleAgent.Po,
    RoleAgent.User,
    RoleAgent.Designer,
    RoleAgent.Architect,
    RoleAgent.Security,
  ],
};

  S: 1,
  M: 1,
  L: 2,
};
export const ALL_AGENTS: readonly string[] = [
  COUNCIL_AGENT,
  ...ROLES_BY_COMPLEXITY.L,

export const CouncilErrorCode = {
  TokenRequired: "cli.council.token_required",
  AgentNotEnabled: "cli.council.agent_not_enabled",
  OutputDirectoryNotEmpty: "cli.council.output_directory_not_empty",
  AgentSessionFailed: "cli.council.agent_session_failed",
  AnswerInvalid: "cli.council.answer_invalid",
  PlanFileInvalid: "cli.council.plan_file_invalid",
} as const;

export const triageSchema = z.strictObject({
  complexity: complexitySchema,
  reasons: z.array(z.string().min(1)).min(1),
});
export type Triage = z.infer<typeof triageSchema>;

export const questionsSchema = z.strictObject({
  questions: z.array(z.string().min(1)).min(1),
});
export type Questions = z.infer<typeof questionsSchema>;

export const roleAnswerSchema = z.strictObject({
  answers: z
    .array(
      z.strictObject({
        question: z.string().min(1),
        answer: z.string().min(1),
        confidence: z.enum(["low", "medium", "high"]),
        assumptions: z.array(z.string().min(1)),
      }),

    .min(1),
});

export const synthesisSchema = z.strictObject({
  agreements: z.array(z.string().min(1)),
  conflicts: z.array(z.string().min(1)),
  direction: z.string().min(1),
});

export const planEntrySchema = importEntrySchema
  .omit({ id: true })
  .extend({ bindings: z.array(z.string()).length(0) });
export type PlanEntry = z.infer<typeof planEntrySchema>;

export const planSchema = z.strictObject({
  entries: z.array(planEntrySchema).min(1),
  reasons: z.record(z.string(), z.string().min(1)),
});
export type Plan = z.infer<typeof planSchema>;

export interface Round {
  questions: string[];
  answers: Partial<Record<RoleAgent, RoleAnswer>>;
}

export interface CouncilRecord {
  idea: string;
  triage: Triage;
  rounds: Round[];
  synthesis: Synthesis;
  plan: Plan;
```

````ts
import { z } from "zod";
import { Diagnostic } from "../../../kernel/errors.ts";
import { CouncilErrorCode } from "./contract.ts";

const JSON_FENCE = /```json[^\n]*\n([\s\S]*?)\n```/g;
const FENCE_BODY = 1;

export function parseAnswer<S extends z.ZodType>(
  text: string,
  schema: S,
  step: string,
): z.infer<S> {
  const fences = [...text.matchAll(JSON_FENCE)];
  const source =
    fences.length > 0 ? fences[fences.length - 1]![FENCE_BODY]! : text.trim();
  let value: unknown;
  try {
    value = JSON.parse(source);
  } catch {
    throw new Diagnostic(
      CouncilErrorCode.AnswerInvalid,
      `${step}: the answer holds no JSON value.`,
    );
  }
  const result = schema.safeParse(value);
  if (!result.success)
    throw new Diagnostic(
      CouncilErrorCode.AnswerInvalid,
      `${step}: ${z.prettifyError(result.error)}`,
    );
  return result.data;
}
````

```ts
import type { Round, Synthesis, Triage } from "./contract.ts";

function shape(value: unknown): string {
  return `Answer with one \`\`\`json block of this shape:\n${JSON.stringify(value, null, 2)}`;
}

function context(idea: string, triage?: Triage): string {
  const lines = [`## Idea\n\n${idea}`];
  if (triage)
    lines.push(
      `## Triage\n\nComplexity ${triage.complexity}.\n${triage.reasons.map((reason) => `- ${reason}`).join("\n")}`,
    );
  return lines.join("\n\n");
}

function questionList(questions: readonly string[]): string {
  return `## Questions\n\n${questions.map((question, index) => `${index + 1}. ${question}`).join("\n")}`;
}

export function triageMessage(idea: string): string {
  return [
    "Step: triage.",
    context(idea),
    shape({ complexity: "S | M | L", reasons: ["..."] }),
  ].join("\n\n");
}

export function questionsMessage(idea: string, triage: Triage): string {
  return [
    "Step: questions. Ask the open questions that the roles must answer.",
    context(idea, triage),
    shape({ questions: ["..."] }),
  ].join("\n\n");
}

export function roleMessage(
  idea: string,
  triage: Triage,
  questions: readonly string[],
): string {
  return [
    "Step: answer every question from your role.",
    context(idea, triage),
    questionList(questions),
    shape({
      answers: [
        {
          question: "...",
          answer: "...",
          confidence: "low | medium | high",
          assumptions: ["..."],
        },
      ],
    }),
  ].join("\n\n");
}

export function conflictQuestionsMessage(
  idea: string,
  triage: Triage,
  round: Round,
): string {
  return [
    "Step: conflict questions. Ask only about the points where the roles below disagree.",
    context(idea, triage),
    `## Answers\n\n${JSON.stringify(round.answers, null, 2)}`,
    shape({ questions: ["..."] }),
  ].join("\n\n");
```

```ts
import { parseAnswer } from "./answer.ts";
import {
  COUNCIL_AGENT,
  planSchema,
  questionsSchema,
  roleAnswerSchema,
  ROLES_BY_COMPLEXITY,
  ROUNDS_BY_COMPLEXITY,
  synthesisSchema,
  triageSchema,
  type CouncilRecord,
  type Round,
  type Triage,
} from "./contract.ts";
import {
  conflictQuestionsMessage,
  planMessage,
  questionsMessage,
  roleMessage,
  synthesisMessage,
  triageMessage,
} from "./messages.ts";

export type Ask = (agentName: string, text: string) => Promise<string>;

const SECOND_ROUND = 2;

async function panel(
  idea: string,
  triage: Triage,
  questions: string[],
  ask: Ask,
): Promise<Round> {
  const roles = ROLES_BY_COMPLEXITY[triage.complexity];
  const answers = await Promise.all(
    roles.map(async (role) =>
      parseAnswer(
        await ask(role, roleMessage(idea, triage, questions)),
        roleAnswerSchema,
        role,
      ),
    ),
  );
  return {
    questions,
    answers: Object.fromEntries(
      roles.map((role, index) => [role, answers[index]!]),
    ),
  };
}

export async function runCouncil(
  idea: string,
  ask: Ask,
  onStep: (step: string) => void = () => {},
): Promise<CouncilRecord> {
  onStep("triage");
  const triage = parseAnswer(
    await ask(COUNCIL_AGENT, triageMessage(idea)),
    triageSchema,
    "triage",
  );
  onStep("questions");
  const { questions } = parseAnswer(
    await ask(COUNCIL_AGENT, questionsMessage(idea, triage)),
    questionsSchema,
    "questions",
  );
  onStep(`round 1: ${ROLES_BY_COMPLEXITY[triage.complexity].join(", ")}`);
  const rounds = [await panel(idea, triage, questions, ask)];
  if (ROUNDS_BY_COMPLEXITY[triage.complexity] >= SECOND_ROUND) {
    onStep("conflict questions");
    const conflicts = parseAnswer(
      await ask(
        COUNCIL_AGENT,
        conflictQuestionsMessage(idea, triage, rounds[0]!),
      ),
      questionsSchema,
      "conflict questions",
    );
    onStep("round 2");
    rounds.push(await panel(idea, triage, conflicts.questions, ask));
  }
  onStep("synthesis");
  const synthesis = parseAnswer(
    await ask(COUNCIL_AGENT, synthesisMessage(idea, triage, rounds)),
    synthesisSchema,
    "synthesis",
  );
  onStep("plan");
  const plan = parseAnswer(
    await ask(COUNCIL_AGENT, planMessage(idea, synthesis)),
    planSchema,
    "plan",
  );
  return { idea, triage, rounds, synthesis, plan };
}
```

```ts
import { CodedError, Diagnostic } from "../../../kernel/errors.ts";
import { httpClient } from "../../../gateway/client.ts";
import {
  OperationResultType,
  type OperationResult,
} from "../../../kernel/operation.ts";
import { isString } from "../../../kernel/values.ts";
import {
  workbenchOperations,
  type SessionEntry,
  type SessionEvents,
  type WorkbenchConfiguration,
} from "../../../workbench/contract.ts";
import { CouncilErrorCode } from "./contract.ts";

export const SESSION_DEADLINE_MS = 10 * 60 * 1000;
const MESSAGE_ENTRY = "message";
const ASSISTANT_ROLE = "assistant";

export interface WorkbenchPort {
  create(
    agentName: string,
    configuration: WorkbenchConfiguration,
  ): Promise<string>;
  message(sessionId: string, text: string): Promise<void>;
  events(
    sessionId: string,
    version: number | undefined,
  ): Promise<SessionEvents>;
  entries(sessionId: string): Promise<SessionEntry[]>;
  abort(sessionId: string): Promise<void>;
}

function failed(agentName: string, reason: string): Diagnostic {
  return new Diagnostic(
    CouncilErrorCode.AgentSessionFailed,
    `${agentName}: ${reason}`,
  );
}

function lastAssistantText(
  entries: readonly SessionEntry[],
): string | undefined {
  const texts = entries
    .filter((entry) => entry.type === MESSAGE_ENTRY)
    .map((entry) => entry.message as { role: string; content: unknown })
    .filter((message) => message.role === ASSISTANT_ROLE)
    .map((message) =>
      isString(message.content)
        ? message.content
        : (message.content as { type: string; text?: string }[])
            .map((block) => block.text ?? "")
            .join(""),
    )
    .filter((text) => text.trim() !== "");
  return texts.at(-1);
}

async function waitForEnd(
  port: WorkbenchPort,
  agentName: string,
  sessionId: string,
  now: () => number,
): Promise<void> {
  const deadline = now() + SESSION_DEADLINE_MS;
  let version: number | undefined;
  for (;;) {
    const { snapshot, version: next } = await port.events(sessionId, version);
    if (snapshot.error_message) throw failed(agentName, snapshot.error_message);
    if (snapshot.pending_approval)
      throw failed(agentName, "the run waits for an approval");
    if (!snapshot.run_active) return;
    if (now() >= deadline)
      throw failed(agentName, "the run did not end within 10 minutes");
    version = next;
  }
}

export async function askInSession(
  port: WorkbenchPort,
  agentName: string,
  configuration: WorkbenchConfiguration,
  text: string,
  now: () => number = Date.now,
): Promise<string> {
  let sessionId: string | undefined;
  try {
    sessionId = await port.create(agentName, configuration);
    await port.message(sessionId, text);
    await waitForEnd(port, agentName, sessionId, now);
    const answer = lastAssistantText(await port.entries(sessionId));
    if (answer === undefined)
      throw failed(agentName, "the run wrote no answer");
    return answer;
  } catch (error) {
    if (sessionId !== undefined) await port.abort(sessionId).catch(() => {});
    if (
      error instanceof CodedError &&
      error.code === CouncilErrorCode.AgentSessionFailed
    )
      throw error;
    throw failed(
      agentName,
      error instanceof Error ? error.message : String(error),
    );
  }
}

function completed<T>(result: OperationResult<T>, operation: string): T {
  if (result.type === OperationResultType.Completed) return result.data;
  if (result.type === OperationResultType.Failure)
    throw new Error(
      `${operation} failed with ${result.error.error.code} (HTTP ${result.status})`,
    );
  throw new Error(`${operation} is indeterminate`);
}

export function httpWorkbenchPort(
  endpoint: string,
  token: string,
): WorkbenchPort {
  const client = httpClient(workbenchOperations, endpoint, token);
  return {
    async create(agentName, configuration) {
      const result = await client["session.create"]({
        params: {},
        query: {},
        body: { agent_name: agentName, ...configuration },
      });
      return completed(result, "workbench.session.create").id;
    },
    async message(sessionId, text) {
      completed(
        await client["session.message"]({
          params: { session_id: sessionId },
          query: {},
          body: { text },
        }),
        "workbench.session.message",
      );
    },
    async events(sessionId, version) {
      return completed(
        await client["session.events"]({
          params: { session_id: sessionId },
          query: version === undefined ? {} : { version },
          body: null,
        }),
        "workbench.session.events",
      );
    },
    async entries(sessionId) {
      return completed(
        await client["session.get"]({
          params: { session_id: sessionId },
          query: {},
          body: null,
        }),
        "workbench.session.get",
      ).entries;
    },
    async abort(sessionId) {
      completed(
        await client["session.abort"]({
          params: { session_id: sessionId },
          query: {},
          body: null,
        }),
        "workbench.session.abort",
      );
    },
  };
}
```

```ts
import { agentOperations, AgentErrorCode } from "../../../agent/contract.ts";
import { httpClient } from "../../../gateway/client.ts";
import { Diagnostic } from "../../../kernel/errors.ts";
import { OperationResultType } from "../../../kernel/operation.ts";
import type { WorkbenchConfiguration } from "../../../workbench/contract.ts";
import { ALL_AGENTS, CouncilErrorCode } from "./contract.ts";

const ENABLED = "enabled";

export type EnablementView = {
  state: "enabled" | "disabled";
  default_configuration: WorkbenchConfiguration;
};
export type EnablementReader = (
  agentName: string,
) => Promise<EnablementView | undefined>;

export async function requireEnabled(
  read: EnablementReader,
  agents: readonly string[] = ALL_AGENTS,
): Promise<Map<string, WorkbenchConfiguration>> {
  const views = await Promise.all(agents.map((agentName) => read(agentName)));
  const missing = agents.filter((_, index) => views[index]?.state !== ENABLED);
  if (missing.length > 0)
    throw new Diagnostic(
      CouncilErrorCode.AgentNotEnabled,
      `enable these agents first: ${missing.join(", ")}`,
    );
  return new Map(
    agents.map((agentName, index) => [
      agentName,
      views[index]!.default_configuration,
    ]),
  );
}

export function httpEnablementReader(
  endpoint: string,
  token: string,
): EnablementReader {
  const client = httpClient(agentOperations, endpoint, token);
  return async (agentName) => {
    const result = await client["enablement.get"]({
      params: { agent_name: agentName },
      query: {},
      body: null,
    });
    if (result.type === OperationResultType.Completed) return result.data;
    if (
      result.type === OperationResultType.Failure &&
      result.error.error.code === AgentErrorCode.NotFound
    )
      return undefined;
    throw new Diagnostic(
      result.type === OperationResultType.Failure
        ? result.error.error.code
        : CouncilErrorCode.AgentSessionFailed,
      `agent.enablement.get ${agentName} failed`,
    );
  };
}
```

```ts
import {
  existsSync,
  mkdirSync,
  readdirSync,
  statSync,
  writeFileSync,
} from "node:fs";
import { join } from "node:path";
import { isDeepStrictEqual } from "node:util";
import { Diagnostic } from "../../../kernel/errors.ts";
import {
  NodeKind,
  planFileNameSchema,
  type PlanFileEntry,
} from "../../../mission/contract.ts";
import { serializePlanFile } from "../../../mission/export.ts";
import { parsePlanFile } from "../../../mission/parser.ts";
import {
  CouncilErrorCode,
  type CouncilRecord,
  type Plan,
  type PlanEntry,
} from "./contract.ts";

export const RECORD_FILENAME = "council.md";
const PARENT_KIND: Readonly<Record<string, string>> = {
  [NodeKind.Objective]: NodeKind.Initiative,
  [NodeKind.Task]: NodeKind.Objective,
};

function invalid(filename: string, reason: string): Diagnostic {
  return new Diagnostic(
    CouncilErrorCode.PlanFileInvalid,
    `${filename}: ${reason}`,
  );
}

function normalized(entry: PlanEntry): PlanEntry {
  return {
    ...entry,
    name: entry.name.trim(),
    requirement: entry.requirement.trim(),
    criterion: entry.criterion.trim(),
  };
}

function checkSet(entries: readonly PlanEntry[]): void {
  const kinds = new Map<string, string>();
  for (const entry of entries) {
    if (!planFileNameSchema.safeParse(entry.filename).success)
      throw invalid(entry.filename, "the filename is not lower-case-words.md");
    if (kinds.has(entry.filename))
      throw invalid(entry.filename, "the filename repeats");
    kinds.set(entry.filename, entry.kind);
  }
  for (const entry of entries) {
    if (entry.kind === NodeKind.Task && entry.depends_on !== undefined)
      throw invalid(entry.filename, "a task holds no depends_on");
    if (
      entry.parent !== undefined &&
      kinds.get(entry.parent) !== PARENT_KIND[entry.kind]
    )
      throw invalid(
        entry.filename,
        `the parent ${entry.parent} is no ${PARENT_KIND[entry.kind] ?? "parent"} of the set`,
      );
    for (const name of entry.depends_on ?? [])
      if (!kinds.has(name))
        throw invalid(
          entry.filename,
          `depends_on names ${name}, which is not in the set`,
        );
  }
}

export function renderPlanFiles(plan: Plan): PlanFileEntry[] {
  const entries = plan.entries.map(normalized);
  checkSet(entries);
  return entries.map((entry) => {
    const content = serializePlanFile(entry);
    let parsed;
    try {
      parsed = parsePlanFile(entry.filename, content);
    } catch (error) {
      throw invalid(
        entry.filename,
        error instanceof Error ? error.message : String(error),
      );
    }
    const expected = { ...entry, depends_on: entry.depends_on ?? [] };
    if (!isDeepStrictEqual(parsed, expected))
      throw invalid(
        entry.filename,
        "the plan file does not read back unchanged",
      );
    return { filename: entry.filename, content };
  });
}

function bullets(items: readonly string[]): string {
  return items.length === 0
    ? "- None."
    : items.map((item) => `- ${item}`).join("\n");
}

export function renderRecord(record: CouncilRecord): string {
  const parts = [
    "# Council record",
    `## Idea\n\n${record.idea}`,
    `## Triage\n\nComplexity ${record.triage.complexity}.\n\n${bullets(record.triage.reasons)}`,
  ];
  record.rounds.forEach((round, index) => {
    parts.push(
      `## Round ${index + 1}\n\n### Questions\n\n${bullets(round.questions)}`,
    );
    for (const [role, answer] of Object.entries(round.answers)) {
      const lines = answer!.answers.map(
        (item) =>
          `- **${item.question}** ${item.answer} (confidence ${item.confidence}; assumptions: ${item.assumptions.length === 0 ? "none" : item.assumptions.join("; ")})`,
      );
      parts.push(`### ${role}\n\n${lines.join("\n")}`);
    }
  });
  parts.push(
    `## Synthesis\n\n### Agreements\n\n${bullets(record.synthesis.agreements)}\n\n### Conflicts\n\n${bullets(record.synthesis.conflicts)}\n\n### Direction\n\n${record.synthesis.direction}`,
  );
  parts.push(
    `## Plan\n\n${record.plan.entries
      .map(
        (entry) =>
          `- \`${entry.filename}\` (${entry.kind}): ${record.plan.reasons[entry.filename] ?? "No reason given."}`,
      )
      .join("\n")}`,
  );
  return `${parts.join("\n\n")}\n`;
}

export function requireEmptyOutput(dir: string): void {
  if (!existsSync(dir)) return;
  if (!statSync(dir).isDirectory() || readdirSync(dir).length > 0)
    throw new Diagnostic(
      CouncilErrorCode.OutputDirectoryNotEmpty,
      `${dir} exists and is not an empty directory`,
    );
}

export function writeOutput(
  dir: string,
  files: readonly PlanFileEntry[],
  record: string,
): void {
  requireEmptyOutput(dir);
  mkdirSync(dir, { recursive: true });
  for (const file of files)
    writeFileSync(join(dir, file.filename), file.content);
  writeFileSync(join(dir, RECORD_FILENAME), record);
}
```

`parsePlanFile` returns no `id` key for a file without an id and always returns `depends_on`, so `expected` adds `depends_on: []` for a task before the comparison.

```ts
import assert from "node:assert/strict";
import { resolve } from "node:path";
import { Command } from "commander";
import { resolveClient } from "../../gateway/client.ts";
import { CommandName, PROGRAM_NAME } from "./constants.ts";
import { requireToken, singleUse } from "./shared.ts";
import { CouncilErrorCode } from "./council/contract.ts";
import { httpEnablementReader, requireEnabled } from "./council/enablement.ts";
import {
  renderPlanFiles,
  renderRecord,
  requireEmptyOutput,
  writeOutput,
} from "./council/output.ts";
import { runCouncil } from "./council/run.ts";
import { askInSession, httpWorkbenchPort } from "./council/session.ts";

async function council(idea: string, command: Command): Promise<void> {
  const options = command.optsWithGlobals();
  const { endpoint, token } = resolveClient(options);
  requireToken(token, CouncilErrorCode.TokenRequired);
  const out = resolve(options.out);
  requireEmptyOutput(out);
  const configurations = await requireEnabled(
    httpEnablementReader(endpoint, token),
  );
  const port = httpWorkbenchPort(endpoint, token);
  const record = await runCouncil(
    idea,
    (agentName, text) =>
      askInSession(port, agentName, configurations.get(agentName)!, text),
    (step) => process.stderr.write(`council: ${step}\n`),
  );
  const files = renderPlanFiles(record.plan);
  writeOutput(out, files, renderRecord(record));
  process.stdout.write(
    `Wrote ${files.length} plan files and council.md to ${out}\n`,
  );
}

export function addCouncilCommand(program: Command): void {
  assert.equal(program.name(), PROGRAM_NAME);
  assert.ok(
    !program.commands.some((command) => command.name() === CommandName.Council),
  );
  program
    .command(CommandName.Council)
    .description(
      "Ask the council agents for a development direction and write plan files",
    )
    .argument("<idea>", "The raw idea")
    .requiredOption(
      "--out <dir>",
      "Absent or empty directory for the plan files and council.md",
    )
    .option("--endpoint <url>", "Server endpoint")
    .option(
      "--token <token>",
      "Caller JWT (otherwise KANTHORD_TOKEN or cli.yaml)",
      singleUse("--token"),
    )
    .action(async (idea: string, _options: unknown, command: Command) =>
      council(idea, command),
    );
}
```
